 def _extract_json(text: str) -> dict:
     text = text.strip()
     if text.startswith("```"):
         text = text.strip("`")
         if text.startswith("json"):
             text = text[4:]
     start, end = text.find("{"), text.rfind("}")
     if start < 0 or end < start:
         raise ValueError("模型未返回 JSON")
     return json.loads(text[start : end + 1])
